// Samples points on the surface of the Lite3 (display meshes, standing pose from MuJoCo) for the diffusion article.
// Run: cp this file to tests/ml/zz-dog.test.ts, `pnpm vitest run tests/ml/zz-dog`, delete the copy.
import { readFileSync, writeFileSync } from "node:fs";
import { readFile } from "node:fs/promises";
import { it } from "vitest";
import { Lite3Sim } from "@/content/posts/lite3-walking/components/sim";

const N = 8000;
const PARTS: ({ mesh: string; mirror?: boolean } | null)[] = [null, { mesh: "torso" }];
for (const side of ["FL", "FR", "HL", "HR"]) PARTS.push({ mesh: "hip" }, { mesh: "thigh", mirror: side[1] === "R" }, { mesh: "shank" }, null);

function triangles(name: string): Float32Array {
  const b = readFileSync(`public/lite3/visual/${name}.bin`), n = b.readUInt32LE(80), out = new Float32Array(n * 9);
  for (let i = 0; i < n; i++) for (let k = 0; k < 9; k++) out[i * 9 + k] = b.readFloatLE(84 + i * 50 + 12 + k * 4);
  return out;
}
const rotate = ([w, x, y, z]: number[], [a, b, c]: number[]) => {
  const tx = 2 * (y * c - z * b), ty = 2 * (z * a - x * c), tz = 2 * (x * b - y * a);
  return [a + w * tx + (y * tz - z * ty), b + w * ty + (z * tx - x * tz), c + w * tz + (x * ty - y * tx)];
};

it("dog points", async () => {
  const sim = await Lite3Sim.load(async (p) => new Uint8Array(await readFile(`public/lite3/${p}`)));
  sim.knobs.command = [0, 0, 0]; sim.reset(); sim.advance(1500); // let it settle into its stance
  let seed = 7; const rand = () => { seed = (seed * 1664525 + 1013904223) >>> 0; return seed / 2 ** 32; };
  const tris: { v: number[][]; area: number; boost: number }[] = [];
  PARTS.forEach((part, id) => {
    if (!part) return;
    const t = triangles(part.mesh), pos = Array.from(sim.bodies.subarray(id * 3, id * 3 + 3)), q = Array.from(sim.orientations.subarray(id * 4, id * 4 + 4));
    for (let i = 0; i < t.length; i += 9) {
      const v = [0, 3, 6].map((o) => rotate(q, [t[i + o], t[i + o + 1] * (part.mirror ? -1 : 1), t[i + o + 2]]).map((c, k) => c + pos[k]));
      const e1 = v[1].map((c, k) => c - v[0][k]), e2 = v[2].map((c, k) => c - v[0][k]);
      tris.push({ v, area: Math.hypot(e1[1] * e2[2] - e1[2] * e2[1], e1[2] * e2[0] - e1[0] * e2[2], e1[0] * e2[1] - e1[1] * e2[0]) / 2, boost: part.mesh === "torso" ? 1 : 2.5 });
    }
  });
  // Thin parts (legs) hold little area; weight each part so the torso gets about half of the points, not four fifths.
  const total = tris.reduce((s, t) => s + t.area * t.boost, 0), cum: number[] = []; let acc = 0;
  for (const t of tris) cum.push((acc += (t.area * t.boost) / total));
  const pts = new Float64Array(N * 3);
  for (let i = 0; i < N; i++) {
    const r = rand(); let lo = 0, hi = cum.length - 1; while (lo < hi) { const m = (lo + hi) >> 1; if (cum[m] < r) lo = m + 1; else hi = m; }
    let a = rand(), b = rand(); if (a + b > 1) { a = 1 - a; b = 1 - b; }
    const { v } = tris[lo]; for (let k = 0; k < 3; k++) pts[i * 3 + k] = v[0][k] + a * (v[1][k] - v[0][k]) + b * (v[2][k] - v[0][k]);
  }
  const lo = [0, 1, 2].map((k) => Math.min(...Array.from({ length: N }, (_, i) => pts[i * 3 + k]))), hi = [0, 1, 2].map((k) => Math.max(...Array.from({ length: N }, (_, i) => pts[i * 3 + k])));
  const half = Math.max(...lo.map((l, k) => (hi[k] - l) / 2)), q16 = new Int16Array(N * 3);
  for (let i = 0; i < N; i++) for (let k = 0; k < 3; k++) q16[i * 3 + k] = Math.round((((pts[i * 3 + k] - (lo[k] + hi[k]) / 2) / half) * 0.95) * 32767);
  writeFileSync("public/diffusion/lite3-points.i16", Buffer.from(q16.buffer));
  console.log(`DOG triangles ${tris.length}, extent ${hi.map((h, k) => (h - lo[k]).toFixed(2)).join(" x ")} m, wrote ${q16.byteLength} bytes`);
}, 120_000);
