import { it } from "vitest";
import { Adam, Mat, Tape } from "@/lib/ml";

// ---- a 3-D shape to learn: a trefoil knot tube, points on its surface, scaled into about [-1, 1]
function knot(rand: () => number): [number, number, number] {
  const t = rand() * 2 * Math.PI, a = rand() * 2 * Math.PI, r = 0.18;
  const c = (s: number): [number, number, number] => [Math.sin(s) + 2 * Math.sin(2 * s), Math.cos(s) - 2 * Math.cos(2 * s), -Math.sin(3 * s)];
  const p = c(t), q = c(t + 1e-3), tan = [q[0] - p[0], q[1] - p[1], q[2] - p[2]];
  const n1 = norm(cross(tan, [0.3, 0.5, 0.8])), n2 = norm(cross(tan, n1));
  return [0, 1, 2].map((k) => (p[k] + r * 3 * (Math.cos(a) * n1[k] + Math.sin(a) * n2[k])) / 3) as [number, number, number];
}
const cross = (a: number[], b: number[]) => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const norm = (v: number[]) => { const l = Math.hypot(...v); return v.map((x) => x / l); };
function mulberry(seed: number) { return () => { seed |= 0; seed = (seed + 0x6d2b79f5) | 0; let t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const gauss = (rand: () => number) => Math.sqrt(-2 * Math.log(1 - rand())) * Math.cos(2 * Math.PI * rand());

// ---- DDPM pieces
const T = 100;
const abar = Array.from({ length: T + 1 }, (_, t) => Math.cos(((t / T + 0.008) / 1.008) * Math.PI / 2) ** 2).map((v, _, a) => v / a[0]);
const FREQ = [1, 2, 4, 8], XF = [1, 2, 4];
const IN = 3 + 3 * 2 * XF.length + 2 * FREQ.length;
function features(x: ArrayLike<number>, n: number, ts: ArrayLike<number>): Mat {
  const m = new Mat(n, IN);
  for (let i = 0; i < n; i++) {
    let o = i * IN;
    for (let k = 0; k < 3; k++) m.data[o++] = x[i * 3 + k];
    for (const f of XF) for (let k = 0; k < 3; k++) { m.data[o++] = Math.sin(f * Math.PI * x[i * 3 + k]); m.data[o++] = Math.cos(f * Math.PI * x[i * 3 + k]); }
    for (const f of FREQ) { m.data[o++] = Math.sin((f * Math.PI * ts[i]) / T); m.data[o++] = Math.cos((f * Math.PI * ts[i]) / T); }
  }
  return m;
}
function chamfer(a: Float64Array, b: Float64Array) {
  const one = (p: Float64Array, q: Float64Array) => { let s = 0; for (let i = 0; i < p.length; i += 3) { let best = Infinity; for (let j = 0; j < q.length; j += 3) { const d = (p[i] - q[j]) ** 2 + (p[i + 1] - q[j + 1]) ** 2 + (p[i + 2] - q[j + 2]) ** 2; if (d < best) best = d; } s += Math.sqrt(best); } return s / (p.length / 3); };
  return (one(a, b) + one(b, a)) / 2;
}

it("3-D point diffusion: how fast, how good", () => {
  for (const H of [64, 128]) {
    const rand = mulberry(1), init = (r: number, c: number) => { const m = new Mat(r, c); for (let i = 0; i < m.data.length; i++) m.data[i] = gauss(rand) * Math.sqrt(2 / r); return m; };
    const P: Record<string, Mat> = { w1: init(IN, H), b1: new Mat(1, H), w2: init(H, H), b2: new Mat(1, H), w3: init(H, H), b3: new Mat(1, H), w4: init(H, 3), b4: new Mat(1, 3) };
    const net = (tape: Tape, x: Mat) => { let h = tape.relu(tape.addRow(tape.matmul(x, P.w1), P.b1)); h = tape.relu(tape.addRow(tape.matmul(h, P.w2), P.b2)); h = tape.relu(tape.addRow(tape.matmul(h, P.w3), P.b3)); return tape.addRow(tape.matmul(h, P.w4), P.b4); };
    const adam = new Adam(P), B = 256;
    const sample = (n: number, steps: number) => {
      const x = new Float64Array(n * 3).map(() => gauss(rand)); const t0 = performance.now();
      const seq = Array.from({ length: steps }, (_, k) => Math.round(T - (k * T) / steps));
      seq.forEach((t, k) => {
        const tPrev = k + 1 < steps ? seq[k + 1] : 0, eps = net(new Tape(), features(x, n, new Float64Array(n).fill(t))).data;
        for (let i = 0; i < n * 3; i++) { const x0 = (x[i] - Math.sqrt(1 - abar[t]) * eps[i]) / Math.sqrt(abar[t]); x[i] = Math.sqrt(abar[tPrev]) * Math.max(-1.5, Math.min(1.5, x0)) + Math.sqrt(1 - abar[tPrev]) * eps[i]; } // DDIM, eta 0
      });
      return { x, ms: performance.now() - t0 };
    };
    const truth = new Float64Array(1500 * 3); for (let i = 0; i < 1500; i++) truth.set(knot(rand), i * 3);
    const self = new Float64Array(1500 * 3); for (let i = 0; i < 1500; i++) self.set(knot(rand), i * 3);
    console.log(`DIFF H=${H} params=${Object.values(P).reduce((n, m) => n + m.data.length, 0)} in=${IN}  chamfer floor (data vs data) ${chamfer(truth, self).toFixed(4)}`);
    const t0 = performance.now(); let ema = 0;
    for (let step = 1; step <= 6000; step++) {
      const x = new Float64Array(B * 3), eps = new Float64Array(B * 3), ts = new Float64Array(B);
      for (let i = 0; i < B; i++) { const p = knot(rand), t = 1 + Math.floor(rand() * T); ts[i] = t; for (let k = 0; k < 3; k++) { const e = gauss(rand); eps[i * 3 + k] = e; x[i * 3 + k] = Math.sqrt(abar[t]) * p[k] + Math.sqrt(1 - abar[t]) * e; } }
      for (const m of Object.values(P)) m.grad.fill(0);
      const tape = new Tape(), loss = tape.mse(net(tape, features(x, B, ts)), eps); tape.backward(); adam.step(step < 4000 ? 2e-3 : 5e-4);
      ema = ema ? ema * 0.98 + loss * 0.02 : loss;
      if ([250, 500, 1000, 2000, 4000, 6000].includes(step)) { const s = sample(1500, 25); console.log(`DIFF H=${H} step ${step} ${((performance.now() - t0) / 1000).toFixed(1)}s loss ${ema.toFixed(3)} chamfer ${chamfer(s.x, truth).toFixed(4)} | sampling 1500 pts x 25 steps ${s.ms.toFixed(0)} ms`); }
    }
  }
}, 900_000);
