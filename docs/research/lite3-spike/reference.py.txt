"""Reference rollout: official Lite3 policy in Python MuJoCo on the flat floor of Lite3.xml.
Mirrors run_policy/lite3_test_policy_runner_onnx.hpp and interface/robot/simulation/mujoco_simulation.py.
Writes weights.json (the MLP) and trace.json (obs/action pairs + trajectory) for the JS side to match."""
import json, sys, time
import numpy as np, onnx, onnxruntime as ort, mujoco
from onnx import numpy_helper

XML = "model/Lite3/Lite3_mjcf/mjcf/Lite3.xml"
ONNX = "repo/policy/ppo/policy.onnx"
DT, DECIMATION, KP, KD = 0.001, 12, 30.0, 1.0
DEFAULT = np.array([0.0, -0.65, 1.3] * 4)
ACTION_SCALE = np.array([0.125, 0.25, 0.25] * 4)
cmd = np.array([float(x) for x in sys.argv[1:4]] or [0.5, 0.0, 0.0])  # vx, vy, yaw rate (already × max 0.8 applied by caller)
SECONDS = 10.0

w = {t.name: numpy_helper.to_array(t) for t in onnx.load(ONNX).graph.initializer}
layers = [(w[f"mlp.{i}.weight"].astype(np.float64), w[f"mlp.{i}.bias"].astype(np.float64)) for i in (0, 2, 4, 6)]
def mlp(x):
    for i, (W, b) in enumerate(layers):
        x = W @ x + b
        if i < 3: x = np.where(x > 0, x, np.expm1(x))  # ELU, alpha 1
    return x
sess = ort.InferenceSession(ONNX)
probe = np.random.default_rng(0).normal(size=45).astype(np.float32)
print("numpy MLP vs onnxruntime, max abs diff:", np.abs(mlp(probe.astype(np.float64)) - sess.run(None, {"obs": probe[None]})[0][0]).max())

model = mujoco.MjModel.from_xml_path(XML)
model.opt.timestep = DT
data = mujoco.MjData(model)
print("nq", model.nq, "nv", model.nv, "nu", model.nu, "ngeom", model.ngeom, "nmesh", model.nmesh, "mesh verts", model.nmeshvert)
print("joint order:", [model.joint(i).name for i in range(1, model.njnt)])
data.qpos[:3] = [0, 0, 0.36]
data.qpos[3:7] = [1, 0, 0, 0]
data.qpos[7:] = DEFAULT
mujoco.mj_forward(model, data)

last_action, target = np.zeros(12), DEFAULT.copy()
trace, traj = [], []
steps = int(SECONDS / DT)
t0 = time.perf_counter()
for k in range(steps):
    if k % DECIMATION == 0:
        R = np.zeros(9); mujoco.mju_quat2Mat(R, data.qpos[3:7]); R = R.reshape(3, 3)
        obs = np.concatenate([data.qvel[3:6] * 0.25, R.T @ np.array([0, 0, -1.0]), cmd,
                              data.qpos[7:] - DEFAULT, data.qvel[6:] * 0.05, last_action])
        last_action = mlp(obs)
        target = last_action * ACTION_SCALE + DEFAULT
        if len(trace) < 50: trace.append({"obs": obs.tolist(), "action": last_action.tolist()})
    data.ctrl[:] = KP * (target - data.qpos[7:]) + KD * (0 - data.qvel[6:])
    mujoco.mj_step(model, data)
    if k % 100 == 0: traj.append([data.time, *data.qpos[:3].tolist()])
wall = time.perf_counter() - t0
x, y, z = data.qpos[:3]
print(f"cmd {cmd.tolist()}  after {SECONDS}s: x={x:.2f} y={y:.2f} z={z:.3f}  mean vx={x / SECONDS:.2f} m/s  min z={min(p[3] for p in traj):.3f}")
print(f"wall {wall:.2f}s for {SECONDS}s sim → {SECONDS / wall:.1f}× real time (native)")
json.dump({"layers": [{"w": W.flatten().tolist(), "b": b.tolist(), "rows": W.shape[0], "cols": W.shape[1]} for W, b in layers]}, open("weights.json", "w"))
json.dump({"cmd": cmd.tolist(), "trace": trace, "traj": traj, "final": [x, y, z]}, open("trace.json", "w"))
